\subsection{Real-Time Data Streaming Pipeline}

The system implements a sophisticated streaming architecture for real-time data processing and immediate insight generation.

\begin{lstlisting}[language=Python, caption=Real-Time Data Streaming Implementation]
import asyncio
import json
from typing import Dict, List, Any
from dataclasses import dataclass, asdict
from datetime import datetime
import kafka
import redis
import pandas as pd

@dataclass
class DevelopmentEvent:
    event_type: str
    user_id: str
    project_id: str
    timestamp: datetime
    metadata: Dict[str, Any]
    
class RealTimeDataCollector:
    def __init__(self, kafka_config: Dict, redis_client: redis.Redis):
        self.kafka_producer = kafka.KafkaProducer(
            **kafka_config,
            value_serializer=lambda x: json.dumps(x, default=str).encode()
        )
        self.redis_client = redis_client
        self.event_handlers = {
            'task_created': self._handle_task_creation,
            'task_completed': self._handle_task_completion,
            'commit_pushed': self._handle_code_commit,
            'estimation_requested': self._handle_estimation_request
        }
    
    async def collect_event(self, event: DevelopmentEvent):
        """Collect and process development events in real-time"""
        try:
            # Immediate processing for critical events
            if event.event_type in self.event_handlers:
                await self.event_handlers[event.event_type](event)
            
            # Stream to Kafka for distributed processing
            self.kafka_producer.send(
                topic='development_events',
                value=asdict(event),
                key=f"{event.project_id}:{event.user_id}"
            )
            
            # Cache recent events for quick access
            cache_key = f"recent_events:{event.user_id}"
            self.redis_client.lpush(cache_key, json.dumps(asdict(event), default=str))
            self.redis_client.ltrim(cache_key, 0, 100)  # Keep last 100 events
            self.redis_client.expire(cache_key, 3600)   # Expire after 1 hour
            
        except Exception as e:
            await self._handle_collection_error(event, e)
    
    async def _handle_task_creation(self, event: DevelopmentEvent):
        """Process task creation events for immediate insights"""
        task_data = event.metadata
        
        # Extract task complexity indicators
        complexity_score = await self._calculate_complexity_score(task_data)
        
        # Update user activity patterns
        await self._update_user_patterns(event.user_id, 'task_creation', complexity_score)
        
        # Trigger proactive estimation if configured
        if task_data.get('auto_estimate', False):
            await self._trigger_proactive_estimation(event)
    
    async def _calculate_complexity_score(self, task_data: Dict) -> float:
        """Calculate task complexity based on multiple factors"""
        factors = {
            'description_length': len(task_data.get('description', '')),
            'acceptance_criteria_count': len(task_data.get('acceptance_criteria', [])),
            'labels_count': len(task_data.get('labels', [])),
            'dependencies_count': len(task_data.get('dependencies', [])),
            'technology_stack_complexity': self._assess_tech_complexity(task_data)
        }
        
        # Weighted complexity calculation
        weights = {
            'description_length': 0.2,
            'acceptance_criteria_count': 0.3,
            'labels_count': 0.1,
            'dependencies_count': 0.25,
            'technology_stack_complexity': 0.15
        }
        
        normalized_factors = {
            'description_length': min(factors['description_length'] / 1000, 1.0),
            'acceptance_criteria_count': min(factors['acceptance_criteria_count'] / 10, 1.0),
            'labels_count': min(factors['labels_count'] / 5, 1.0),
            'dependencies_count': min(factors['dependencies_count'] / 8, 1.0),
            'technology_stack_complexity': factors['technology_stack_complexity']
        }
        
        complexity_score = sum(
            normalized_factors[factor] * weights[factor] 
            for factor in weights
        )
        
        return complexity_score
\end{lstlisting}

\subsection{Feature Engineering Framework}

TaskWeight implements an advanced feature engineering pipeline that transforms raw development data into meaningful features for machine learning models.

\begin{lstlisting}[language=Python, caption=Advanced Feature Engineering Pipeline]
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.feature_extraction.text import TfidfVectorizer
from typing import Dict, List, Tuple, Any
import re
from datetime import datetime, timedelta

class AdvancedFeatureEngineer:
    def __init__(self):
        self.scalers = {}
        self.encoders = {}
        self.vectorizers = {}
        
    def engineer_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Comprehensive feature engineering for estimation models"""
        
        # Create a copy to avoid modifying original data
        features_df = df.copy()
        
        # 1. Temporal Features
        features_df = self._create_temporal_features(features_df)
        
        # 2. Text-based Features
        features_df = self._create_text_features(features_df)
        
        # 3. Developer Performance Features
        features_df = self._create_developer_features(features_df)
        
        # 4. Project Context Features
        features_df = self._create_project_features(features_df)
        
        # 5. Technical Complexity Features
        features_df = self._create_complexity_features(features_df)
        
        # 6. Historical Pattern Features
        features_df = self._create_historical_features(features_df)
        
        # 7. Interaction Features
        features_df = self._create_interaction_features(features_df)
        
        return features_df
    
    def _create_temporal_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Create time-based features that capture temporal patterns"""
        
        # Convert timestamp to datetime if needed
        if 'created_at' in df.columns:
            df['created_at'] = pd.to_datetime(df['created_at'])
            
            # Basic temporal features
            df['hour_of_day'] = df['created_at'].dt.hour
            df['day_of_week'] = df['created_at'].dt.dayofweek
            df['month'] = df['created_at'].dt.month
            df['quarter'] = df['created_at'].dt.quarter
            df['is_weekend'] = df['day_of_week'].isin([5, 6]).astype(int)
            df['is_business_hours'] = ((df['hour_of_day'] >= 9) & 
                                     (df['hour_of_day'] <= 17)).astype(int)
            
            # Cyclical encoding for temporal features
            df['hour_sin'] = np.sin(2 * np.pi * df['hour_of_day'] / 24)
            df['hour_cos'] = np.cos(2 * np.pi * df['hour_of_day'] / 24)
            df['day_sin'] = np.sin(2 * np.pi * df['day_of_week'] / 7)
            df['day_cos'] = np.cos(2 * np.pi * df['day_of_week'] / 7)
            df['month_sin'] = np.sin(2 * np.pi * df['month'] / 12)
            df['month_cos'] = np.cos(2 * np.pi * df['month'] / 12)
        
        return df
    
    def _create_text_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Extract features from task descriptions and comments"""
        
        if 'task_description' in df.columns:
            # Basic text statistics
            df['description_length'] = df['task_description'].str.len()
            df['word_count'] = df['task_description'].str.split().str.len()
            df['sentence_count'] = df['task_description'].str.count(r'[.!?]+')
            df['avg_word_length'] = (df['description_length'] / 
                                   df['word_count'].replace(0, 1))
            
            # Technical keyword detection
            technical_keywords = [
                'api', 'database', 'frontend', 'backend', 'algorithm',
                'optimization', 'security', 'performance', 'integration',
                'refactor', 'migration', 'testing', 'deployment'
            ]
            
            for keyword in technical_keywords:
                df[f'has_{keyword}'] = df['task_description'].str.contains(
                    keyword, case=False, na=False
                ).astype(int)
            
            # Complexity indicators
            complexity_indicators = [
                r'\b(complex|difficult|challenging)\b',
                r'\b(multiple|several|various)\b',
                r'\b(integrate|coordination|synchroniz)\w*\b',
                r'\b(performance|optimization|scalability)\b'
            ]
            
            for i, pattern in enumerate(complexity_indicators):
                df[f'complexity_indicator_{i}'] = df['task_description'].str.contains(
                    pattern, case=False, na=False, regex=True
                ).astype(int)
            
            # TF-IDF features for semantic similarity
            if 'task_description' in df.columns and len(df) > 1:
                tfidf = TfidfVectorizer(
                    max_features=100,
                    stop_words='english',
                    ngram_range=(1, 2)
                )
                
                tfidf_matrix = tfidf.fit_transform(df['task_description'].fillna(''))
                tfidf_features = pd.DataFrame(
                    tfidf_matrix.toarray(),
                    columns=[f'tfidf_{i}' for i in range(tfidf_matrix.shape[1])],
                    index=df.index
                )
                df = pd.concat([df, tfidf_features], axis=1)
                self.vectorizers['task_description'] = tfidf
        
        return df
    
    def _create_developer_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Create features based on developer performance history"""
        
        if 'user_id' in df.columns:
            # Historical accuracy by developer
            user_accuracy = df.groupby('user_id')['accuracy_score'].agg([
                'mean', 'std', 'count', 'median'
            ]).add_prefix('user_accuracy_')
            
            # Developer velocity patterns
            user_velocity = df.groupby('user_id')['actual_hours'].agg([
                'mean', 'std', 'median'
            ]).add_prefix('user_velocity_')
            
            # Developer complexity preference
            user_complexity = df.groupby('user_id')['complexity_score'].agg([
                'mean', 'std', 'max'
            ]).add_prefix('user_complexity_')
            
            # Merge developer features
            df = df.merge(user_accuracy, left_on='user_id', right_index=True, how='left')
            df = df.merge(user_velocity, left_on='user_id', right_index=True, how='left')
            df = df.merge(user_complexity, left_on='user_id', right_index=True, how='left')
            
            # Developer experience level (based on historical data)
            df['developer_experience_days'] = (
                df.groupby('user_id')['created_at'].transform('max') -
                df.groupby('user_id')['created_at'].transform('min')
            ).dt.days
            
            df['developer_task_count'] = df.groupby('user_id').cumcount() + 1
        
        return df
    
    def _create_complexity_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Create features that capture task complexity"""
        
        # Technology stack complexity
        if 'technology_stack' in df.columns:
            df['tech_stack_count'] = df['technology_stack'].str.split(',').str.len()
            
            # Known complex technologies
            complex_techs = ['kubernetes', 'microservices', 'machine learning', 
                           'blockchain', 'distributed systems']
            
            for tech in complex_techs:
                df[f'uses_{tech.replace(" ", "_")}'] = df['technology_stack'].str.contains(
                    tech, case=False, na=False
                ).astype(int)
        
        # Dependencies complexity
        if 'dependencies' in df.columns:
            df['dependency_count'] = df['dependencies'].str.split(',').str.len()
            df['has_external_dependencies'] = df['dependencies'].str.contains(
                'external', case=False, na=False
            ).astype(int)
        
        # Priority-based complexity
        if 'priority' in df.columns:
            priority_weights = {'low': 1, 'medium': 2, 'high': 3, 'critical': 4}
            df['priority_weight'] = df['priority'].map(priority_weights).fillna(2)
        
        return df
    
    def _create_interaction_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Create interaction features between different dimensions"""
        
        # User-project interaction
        if 'user_id' in df.columns and 'project_id' in df.columns:
            df['user_project_familiarity'] = df.groupby(['user_id', 'project_id']).cumcount()
        
        # Complexity-priority interaction
        if 'complexity_score' in df.columns and 'priority_weight' in df.columns:
            df['complexity_priority_interaction'] = (
                df['complexity_score'] * df['priority_weight']
            )
        
        # Time-complexity interaction
        if 'is_business_hours' in df.columns and 'complexity_score' in df.columns:
            df['off_hours_complexity'] = (
                (1 - df['is_business_hours']) * df['complexity_score']
            )
        
        return df
\end{lstlisting}

\section{Predictive Analytics Engine}

\subsection{Advanced Forecasting Models}

TaskWeight implements state-of-the-art forecasting models to predict project timelines, resource requirements, and potential bottlenecks.

\begin{lstlisting}[language=Python, caption=Advanced Forecasting Implementation]
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error
import tensorflow as tf
from tensorflow import keras
from typing import List, Dict, Tuple
import warnings
warnings.filterwarnings('ignore')

class AdvancedForecastingEngine:
    def __init__(self):
        self.models = {
            'project_timeline': None,
            'resource_demand': None,
            'bottleneck_prediction': None,
            'quality_forecast': None
        }
        self.feature_importance = {}
        
    def train_project_timeline_forecaster(self, historical_data: pd.DataFrame):
        """Train advanced project timeline forecasting model"""
        
        # Feature engineering for timeline prediction
        features = self._engineer_timeline_features(historical_data)
        
        # Target variable: actual project duration
        target = historical_data['actual_project_duration_hours']
        
        # Create sequence data for LSTM
        X_sequences, y_sequences = self._create_sequences(features, target, seq_length=10)
        
        # Build hybrid model (LSTM + Dense layers)
        model = self._build_timeline_lstm_model(X_sequences.shape[1:])
        
        # Train with early stopping and learning rate scheduling
        early_stopping = keras.callbacks.EarlyStopping(
            monitor='val_loss', patience=10, restore_best_weights=True
        )
        
        lr_scheduler = keras.callbacks.ReduceLROnPlateau(
            monitor='val_loss', factor=0.5, patience=5, min_lr=1e-6
        )
        
        history = model.fit(
            X_sequences, y_sequences,
            batch_size=32,
            epochs=100,
            validation_split=0.2,
            callbacks=[early_stopping, lr_scheduler],
            verbose=0
        )
        
        self.models['project_timeline'] = model
        return history
    
    def _build_timeline_lstm_model(self, input_shape: Tuple) -> keras.Model:
        """Build sophisticated LSTM model for timeline prediction"""
        
        model = keras.Sequential([
            # LSTM layers with dropout for regularization
            keras.layers.LSTM(128, return_sequences=True, input_shape=input_shape),
            keras.layers.Dropout(0.2),
            keras.layers.LSTM(64, return_sequences=True),
            keras.layers.Dropout(0.2),
            keras.layers.LSTM(32, return_sequences=False),
            keras.layers.Dropout(0.2),
            
            # Dense layers for final prediction
            keras.layers.Dense(64, activation='relu'),
            keras.layers.Dropout(0.1),
            keras.layers.Dense(32, activation='relu'),
            keras.layers.Dense(1, activation='linear')
        ])
        
        # Custom optimizer with gradient clipping
        optimizer = keras.optimizers.Adam(
            learning_rate=0.001,
            clipnorm=1.0
        )
        
        model.compile(
            optimizer=optimizer,
            loss='huber',  # Robust to outliers
            metrics=['mae', 'mse']
        )
        
        return model
    
    def predict_project_timeline(self, project_features: Dict) -> Dict:
        """Predict project timeline with confidence intervals"""
        
        if self.models['project_timeline'] is None:
            raise ValueError("Timeline forecasting model not trained")
        
        # Prepare features
        feature_vector = self._prepare_project_features(project_features)
        
        # Generate predictions with Monte Carlo Dropout
        predictions = []
        n_samples = 100
        
        for _ in range(n_samples):
            pred = self.models['project_timeline'](feature_vector, training=True)
            predictions.append(pred.numpy()[0][0])
        
        predictions = np.array(predictions)
        
        # Calculate statistics
        mean_prediction = np.mean(predictions)
        std_prediction = np.std(predictions)
        confidence_95 = np.percentile(predictions, [2.5, 97.5])
        
        return {
            'estimated_hours': mean_prediction,
            'confidence_interval_95': confidence_95,
            'prediction_uncertainty': std_prediction,
            'risk_assessment': self._assess_timeline_risk(mean_prediction, std_prediction)
        }
    
    def train_bottleneck_predictor(self, project_data: pd.DataFrame):
        """Train bottleneck prediction model using ensemble methods"""
        
        # Engineer features for bottleneck detection
        features = self._engineer_bottleneck_features(project_data)
        
        # Multi-label target: different types of bottlenecks
        bottleneck_types = [
            'resource_bottleneck', 'technical_bottleneck', 
            'communication_bottleneck', 'dependency_bottleneck'
        ]
        
        models = {}
        
        for bottleneck_type in bottleneck_types:
            if bottleneck_type in project_data.columns:
                # Binary classification for each bottleneck type
                y = project_data[bottleneck_type].astype(int)
                
                # Use Random Forest with balanced class weights
                model = RandomForestRegressor(
                    n_estimators=200,
                    max_depth=15,
                    min_samples_split=10,
                    min_samples_leaf=5,
                    random_state=42,
                    n_jobs=-1
                )
                
                model.fit(features, y)
                models[bottleneck_type] = model
                
                # Store feature importance
                self.feature_importance[bottleneck_type] = dict(
                    zip(features.columns, model.feature_importances_)
                )
        
        self.models['bottleneck_prediction'] = models
    
    def predict_bottlenecks(self, project_features: Dict) -> Dict:
        """Predict potential bottlenecks with probability scores"""
        
        if self.models['bottleneck_prediction'] is None:
            raise ValueError("Bottleneck prediction model not trained")
        
        feature_vector = self._prepare_bottleneck_features(project_features)
        predictions = {}
        
        for bottleneck_type, model in self.models['bottleneck_prediction'].items():
            probability = model.predict_proba(feature_vector)[0][1]
            risk_level = self._categorize_risk_level(probability)
            
            predictions[bottleneck_type] = {
                'probability': probability,
                'risk_level': risk_level,
                'key_factors': self._get_top_risk_factors(
                    bottleneck_type, feature_vector
                )
            }
        
        return predictions
    
    def generate_optimization_recommendations(self, project_analysis: Dict) -> List[Dict]:
        """Generate actionable optimization recommendations"""
        
        recommendations = []
        
        # Timeline optimization
        if project_analysis.get('timeline_risk', 'low') in ['high', 'critical']:
            recommendations.append({
                'type': 'timeline_optimization',
                'priority': 'high',
                'recommendation': 'Consider parallel task execution and resource reallocation',
                'expected_impact': 'Reduce timeline by 15-25%',
                'implementation_effort': 'medium'
            })
        
        # Resource optimization
        bottlenecks = project_analysis.get('bottlenecks', {})
        if bottlenecks.get('resource_bottleneck', {}).get('risk_level') == 'high':
            recommendations.append({
                'type': 'resource_optimization',
                'priority': 'high',
                'recommendation': 'Add specialized team members or redistribute workload',
                'expected_impact': 'Improve delivery predictability by 30%',
                'implementation_effort': 'high'
            })
        
        # Process optimization
        if bottlenecks.get('communication_bottleneck', {}).get('risk_level') == 'medium':
            recommendations.append({
                'type': 'process_optimization',
                'priority': 'medium',
                'recommendation': 'Implement daily standups and improve documentation',
                'expected_impact': 'Reduce coordination overhead by 20%',
                'implementation_effort': 'low'
            })
        
        return recommendations
